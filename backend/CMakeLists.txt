cmake_minimum_required(VERSION 3.16)
project(webgame_no_named LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Drogon CONFIG REQUIRED)

add_executable(game_server main.cpp stage.cpp)
target_link_libraries(game_server PRIVATE Drogon::Drogon)
include(CTest)
if(BUILD_TESTING)
  add_executable(stage_test stage_test.cpp stage.cpp)
  add_test(NAME stage_replays COMMAND stage_test)
  add_executable(ranking_test ranking_test.cpp)
  target_link_libraries(ranking_test PRIVATE Drogon::Drogon)
  add_test(NAME ranking_scores COMMAND ranking_test)
endif()
